# Sleightbook MVP1 — Part 2: API (Bun + Hono + Drizzle + Postgres)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `apps/api`: Postgres schema + migrations, idempotent Ambitious Card seed, and the 7 MVP1 endpoints (+ health), all tested against a real Postgres test database.

**Architecture:** `createApp(db)` factory returns a chained Hono app mounted at `/api` (so the web app can derive a typed RPC client from `TApp`). Routes validate with Zod schemas from `@sleightbook/shared` via a `zv()` wrapper; services hold all Drizzle queries and scope every query to `getCurrentUserId()`. Errors are thrown as `HttpError` and rendered by one `onError` handler as `{ error: { code, message } }`.

**Tech Stack:** Bun, Hono, @hono/zod-validator, Drizzle ORM + drizzle-kit, postgres (postgres-js), Zod 4, Docker Postgres 16.

**Spec:** [`docs/superpowers/specs/2026-09-26-sleightbook-mvp1-design.md`](../specs/2026-09-26-sleightbook-mvp1-design.md) (§3, §5, §7)

**Series:** [Part 1](2026-09-26-mvp1-part1-foundation-engine.md) → Part 2 (this) → [Part 3: Web & E2E](2026-09-26-mvp1-part3-web.md)

```ts
import { sql } from "drizzle-orm";
import {
  boolean,
  check,
  integer,
  jsonb,
  pgEnum,
  pgTable,
  primaryKey,
  text,
  timestamp,
  unique,
  uuid,
} from "drizzle-orm/pg-core";

import { CATEGORIES, DIFFICULTIES, ITEM_KINDS } from "@sleightbook/shared/schemas/enums";

const id = () => uuid().primaryKey().defaultRandom();
const timestamps = {
  createdAt: timestamp({ withTimezone: true }).notNull().defaultNow(),
  updatedAt: timestamp({ withTimezone: true })
    .notNull()
    .defaultNow()
    .$onUpdate(() => new Date()),
};
const textList = () =>
  text()
    .array()
    .notNull()
    .default(sql`'{}'::text[]`);

export const categoryEnum = pgEnum("category", CATEGORIES);
export const difficultyEnum = pgEnum("difficulty", DIFFICULTIES);
export const itemKindEnum = pgEnum("item_kind", ITEM_KINDS);

export const users = pgTable("users", {
  id: id(),
  name: text().notNull(),
  ...timestamps,
});

export const tricks = pgTable(
  "tricks",
  {
    id: id(),
    userId: uuid()
      .notNull()
      .references(() => users.id, { onDelete: "cascade" }),
    name: text().notNull(),
    slug: text().notNull(),
    description: text().notNull().default(""),
    category: categoryEnum().notNull(),
    difficulty: difficultyEnum().notNull(),
    durationMin: integer().notNull(),
    durationMax: integer().notNull(),
    isFavorite: boolean().notNull().default(false),
    ...timestamps,
  },
  (table) => [unique().on(table.userId, table.slug)],
);

export const routines = pgTable("routines", {
  id: id(),
  trickId: uuid()
    .notNull()
    .references(() => tricks.id, { onDelete: "cascade" }),
  name: text().notNull(),
  description: text().notNull().default(""),
  isDefault: boolean().notNull().default(false),
  position: integer().notNull().default(0),
  tips: textList(),
  ...timestamps,
});

export const phases = pgTable(
  "phases",
  {
    id: id(),
    routineId: uuid()
      .notNull()
      .references(() => routines.id, { onDelete: "cascade" }),
    position: integer().notNull(),
    name: text().notNull(),
    summary: text().notNull().default(""),
    explanation: text().notNull().default(""),
    spectatorText: text().notNull().default(""),
    ...timestamps,
  },
  (table) => [unique().on(table.routineId, table.position)],
);

export const actions = pgTable(
  "actions",
  {
    id: id(),
    phaseId: uuid()
      .notNull()
      .references(() => phases.id, { onDelete: "cascade" }),
    position: integer().notNull(),
    type: text().notNull(),
    params: jsonb().$type<Record<string, unknown>>().notNull(),
    durationMs: integer().notNull().default(700),
    ...timestamps,
  },
  (table) => [unique().on(table.phaseId, table.position)],
);

export const techniques = pgTable("techniques", {
  id: id(),
  userId: uuid()
    .notNull()
    .references(() => users.id, { onDelete: "cascade" }),
  name: text().notNull(),
  description: text().notNull().default(""),
  difficulty: difficultyEnum().notNull(),
  category: text().notNull().default(""),
  tips: textList(),
  commonMistakes: textList(),
  ...timestamps,
});

export const items = pgTable("items", {
  id: id(),
  userId: uuid()
    .notNull()
    .references(() => users.id, { onDelete: "cascade" }),
  kind: itemKindEnum().notNull(),
  name: text().notNull(),
  description: text().notNull().default(""),
  setupNotes: text().notNull().default(""),
  ...timestamps,
});

export const phaseTechniques = pgTable(
  "phase_techniques",
  {
    phaseId: uuid()
      .notNull()
      .references(() => phases.id, { onDelete: "cascade" }),
    techniqueId: uuid()
      .notNull()
      .references(() => techniques.id, { onDelete: "cascade" }),
  },
  (table) => [primaryKey({ columns: [table.phaseId, table.techniqueId] })],
);

export const routineItems = pgTable(
  "routine_items",
  {
    routineId: uuid()
      .notNull()
      .references(() => routines.id, { onDelete: "cascade" }),
    itemId: uuid()
      .notNull()
      .references(() => items.id, { onDelete: "cascade" }),
  },
  (table) => [primaryKey({ columns: [table.routineId, table.itemId] })],
);

export const notes = pgTable(
  "notes",
  {
    id: id(),
    userId: uuid()
      .notNull()
      .references(() => users.id, { onDelete: "cascade" }),
    body: text().notNull(),
    trickId: uuid().references(() => tricks.id, { onDelete: "cascade" }),
    routineId: uuid().references(() => routines.id, { onDelete: "cascade" }),
    phaseId: uuid().references(() => phases.id, { onDelete: "cascade" }),
    techniqueId: uuid().references(() => techniques.id, { onDelete: "cascade" }),
    itemId: uuid().references(() => items.id, { onDelete: "cascade" }),
    ...timestamps,
  },
  () => [
    check(
      "notes_exactly_one_target",
      sql`num_nonnulls(trick_id, routine_id, phase_id, technique_id, item_id) = 1`,
    ),
  ],
);
```

`apps/api/src/db/seed.ts`:

```ts
import { eq } from "drizzle-orm";

import { validateRoutine } from "@sleightbook/engine/timeline";
import { AMBITIOUS_CARD, type ITrickFixture } from "@sleightbook/shared/fixtures/ambitiousCard";

import { requireEnv } from "../env";
import { DEFAULT_USER_ID } from "../services/currentUser";
import { createDb, type TDb } from "./client";
import {
  actions,
  items,
  notes,
  phaseTechniques,
  phases,
  routineItems,
  routines,
  techniques,
  tricks,
  users,
} from "./schema";

export const seed = async (db: TDb, fixture: ITrickFixture = AMBITIOUS_CARD): Promise<void> => {
  const validation = validateRoutine(fixture.phases);
  if (!validation.ok) {
    throw new Error(`Invalid fixture routine: ${validation.error.code} ${validation.error.message}`);
  }

  await db.transaction(async (tx) => {
    await tx.delete(users).where(eq(users.id, DEFAULT_USER_ID));
    await tx.insert(users).values({ id: DEFAULT_USER_ID, name: "Magician" });

    const [trick] = await tx
      .insert(tricks)
      .values({ userId: DEFAULT_USER_ID, ...fixture.trick })
      .returning({ id: tricks.id });

    const [routine] = await tx
      .insert(routines)
      .values({
        trickId: trick.id,
        name: fixture.routine.name,
        description: fixture.routine.description,
        tips: fixture.routine.tips,
        isDefault: true,
        position: 0,
      })
      .returning({ id: routines.id });

    const techniqueRows = await tx
      .insert(techniques)
      .values(fixture.techniques.map((technique) => ({ userId: DEFAULT_USER_ID, ...technique })))
      .returning({ id: techniques.id, name: techniques.name });
    const techniqueIdByName = new Map(techniqueRows.map((row) => [row.name, row.id]));

    if (fixture.items.length > 0) {
      const itemRows = await tx
        .insert(items)
        .values(fixture.items.map((item) => ({ userId: DEFAULT_USER_ID, ...item })))
        .returning({ id: items.id });
      await tx
        .insert(routineItems)
        .values(itemRows.map((item) => ({ routineId: routine.id, itemId: item.id })));
    }

    for (const [position, phase] of fixture.phases.entries()) {
      const [phaseRow] = await tx
        .insert(phases)
        .values({
          routineId: routine.id,
          position,
          name: phase.name,
          summary: phase.summary,
          explanation: phase.explanation,
          spectatorText: phase.spectatorText,
        })
        .returning({ id: phases.id });

      await tx.insert(actions).values(
        phase.actions.map((record, actionPosition) => ({
          phaseId: phaseRow.id,
          position: actionPosition,
          type: record.action.type,
          params: record.action.params,
          durationMs: record.durationMs,
        })),
      );

      const techniqueIds = phase.techniques.map((name) => {
        const techniqueId = techniqueIdByName.get(name);
        if (!techniqueId) throw new Error(`Unknown technique in fixture: ${name}`);
        return techniqueId;
      });
      if (techniqueIds.length > 0) {
        await tx
          .insert(phaseTechniques)
          .values(techniqueIds.map((techniqueId) => ({ phaseId: phaseRow.id, techniqueId })));
      }
    }

    await tx.insert(notes).values({ userId: DEFAULT_USER_ID, trickId: trick.id, body: fixture.trickNote });
```

```ts
import { and, asc, eq, inArray } from "drizzle-orm";

import { ActionSchema } from "@sleightbook/shared/schemas/action";
import type { IRoutineDetail } from "@sleightbook/shared/schemas/routine";

import type { TDb } from "../db/client";
import { actions, notes, phaseTechniques, phases, routines, techniques, tricks } from "../db/schema";
import { notFound } from "../errors";
import { toNoteDto } from "../mappers";
import { getCurrentUserId } from "./currentUser";

export const getRoutineDetail = async (db: TDb, routineId: string): Promise<IRoutineDetail> => {
  const userId = getCurrentUserId();
  const [routine] = await db
    .select({
      id: routines.id,
      trickId: routines.trickId,
      name: routines.name,
      description: routines.description,
      tips: routines.tips,
    })
    .from(routines)
    .innerJoin(tricks, eq(routines.trickId, tricks.id))
    .where(and(eq(routines.id, routineId), eq(tricks.userId, userId)));
  if (!routine) throw notFound("Routine");

  const phaseRows = await db
    .select()
    .from(phases)
    .where(eq(phases.routineId, routineId))
    .orderBy(asc(phases.position));
  const phaseIds = phaseRows.map((phase) => phase.id);
  if (phaseIds.length === 0) return { ...routine, phases: [] };

  const [actionRows, techniqueRows, noteRows] = await Promise.all([
    db.select().from(actions).where(inArray(actions.phaseId, phaseIds)).orderBy(asc(actions.position)),
    db
      .select({ phaseId: phaseTechniques.phaseId, technique: techniques })
      .from(phaseTechniques)
      .innerJoin(techniques, eq(phaseTechniques.techniqueId, techniques.id))
      .where(inArray(phaseTechniques.phaseId, phaseIds))
      .orderBy(asc(techniques.name)),
    db
      .select()
      .from(notes)
      .where(and(inArray(notes.phaseId, phaseIds), eq(notes.userId, userId)))
      .orderBy(asc(notes.createdAt)),
  ]);

  return {
    ...routine,
    phases: phaseRows.map((phase) => ({
      id: phase.id,
      position: phase.position,
      name: phase.name,
      summary: phase.summary,
      explanation: phase.explanation,
      spectatorText: phase.spectatorText,
      actions: actionRows
        .filter((row) => row.phaseId === phase.id)
        .map((row) => ({
          id: row.id,
          position: row.position,
          durationMs: row.durationMs,
          action: ActionSchema.parse({ type: row.type, params: row.params }),
        })),
      techniques: techniqueRows
        .filter((row) => row.phaseId === phase.id)
        .map(({ technique }) => ({
          id: technique.id,
          name: technique.name,
          description: technique.description,
          difficulty: technique.difficulty,
          category: technique.category,
          tips: technique.tips,
          commonMistakes: technique.commonMistakes,
        })),
      notes: noteRows.filter((row) => row.phaseId === phase.id).map(toNoteDto),
    })),
  };
};
```

```ts
import { and, eq } from "drizzle-orm";

import {
  type ICreateNoteInput,
  type INote,
  NOTE_TARGET_KEYS,
  type TNoteTargetKey,
} from "@sleightbook/shared/schemas/note";

import type { TDb } from "../db/client";
import { items, notes, phases, routines, techniques, tricks } from "../db/schema";
import { HttpError, notFound } from "../errors";
import { toNoteDto } from "../mappers";
import { getCurrentUserId } from "./currentUser";

const targetExists = async (db: TDb, key: TNoteTargetKey, targetId: string): Promise<boolean> => {
  const userId = getCurrentUserId();
  switch (key) {
    case "trickId": {
      const rows = await db
        .select({ id: tricks.id })
        .from(tricks)
        .where(and(eq(tricks.id, targetId), eq(tricks.userId, userId)));
      return rows.length > 0;
    }
    case "routineId": {
      const rows = await db
        .select({ id: routines.id })
        .from(routines)
        .innerJoin(tricks, eq(routines.trickId, tricks.id))
        .where(and(eq(routines.id, targetId), eq(tricks.userId, userId)));
      return rows.length > 0;
    }
    case "phaseId": {
      const rows = await db
        .select({ id: phases.id })
        .from(phases)
        .innerJoin(routines, eq(phases.routineId, routines.id))
        .innerJoin(tricks, eq(routines.trickId, tricks.id))
        .where(and(eq(phases.id, targetId), eq(tricks.userId, userId)));
      return rows.length > 0;
    }
    case "techniqueId": {
      const rows = await db
        .select({ id: techniques.id })
        .from(techniques)
        .where(and(eq(techniques.id, targetId), eq(techniques.userId, userId)));
      return rows.length > 0;
    }
    case "itemId": {
      const rows = await db
        .select({ id: items.id })
        .from(items)
        .where(and(eq(items.id, targetId), eq(items.userId, userId)));
      return rows.length > 0;
    }
    default: {
      const unknownKey: never = key;
      throw new Error(`Unknown note target: ${String(unknownKey)}`);
    }
  }
};

export const createNote = async (db: TDb, input: ICreateNoteInput): Promise<INote> => {
  const key = NOTE_TARGET_KEYS.find((candidate) => input[candidate] !== undefined);
  const targetId = key ? input[key] : undefined;
  if (!key || !targetId) throw new HttpError(400, "VALIDATION", "Exactly one note target is required");
  if (!(await targetExists(db, key, targetId))) throw notFound("Note target");

  const [row] = await db
    .insert(notes)
    .values({
      userId: getCurrentUserId(),
      body: input.body,
      trickId: input.trickId ?? null,
      routineId: input.routineId ?? null,
      phaseId: input.phaseId ?? null,
      techniqueId: input.techniqueId ?? null,
      itemId: input.itemId ?? null,
    })
    .returning();
  return toNoteDto(row);
};

export const updateNote = async (db: TDb, noteId: string, body: string): Promise<INote> => {
  const [row] = await db
    .update(notes)
    .set({ body })
    .where(and(eq(notes.id, noteId), eq(notes.userId, getCurrentUserId())))
    .returning();
  if (!row) throw notFound("Note");
  return toNoteDto(row);
};

export const deleteNote = async (db: TDb, noteId: string): Promise<void> => {
  const deleted = await db
    .delete(notes)
    .where(and(eq(notes.id, noteId), eq(notes.userId, getCurrentUserId())))
    .returning({ id: notes.id });
  if (deleted.length === 0) throw notFound("Note");
};
```

```ts
import { Hono } from "hono";

import { CreateNoteSchema, UpdateNoteSchema } from "@sleightbook/shared/schemas/note";

import type { TDb } from "../db/client";
import { createNote, deleteNote, updateNote } from "../services/noteService";
import { IdParamSchema, zv } from "../validate";

export const notesRoutes = (db: TDb) =>
  new Hono()
    .post("/", zv("json", CreateNoteSchema), async (c) =>
      c.json(await createNote(db, c.req.valid("json")), 201),
    )
    .patch("/:id", zv("param", IdParamSchema), zv("json", UpdateNoteSchema), async (c) =>
      c.json(await updateNote(db, c.req.valid("param").id, c.req.valid("json").body)),
    )
    .delete("/:id", zv("param", IdParamSchema), async (c) => {
      await deleteNote(db, c.req.valid("param").id);
      return c.body(null, 204);
    });
```
